# fired at gamestart to setup the habitat
country_event = {
	id = mem_eden_protocol.1
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_origin = origin_mem_eden_protocol
	}
	immediate = {
		set_global_flag = origin_mem_eden_protocol_exists_somewhere
		leader = {
			add_trait = {
				trait = leader_trait_legendary_mem_eden_protocol
				show_message = no
			}
		}
		# This stays for lore reasons, not that anybody will actually use it ^^
		give_technology = {
			tech = tech_terrestrial_sculpting
			message = no
		}
		# The building does the same 
		give_technology = {
			tech = tech_climate_restoration
			message = no
		}
		set_country_flag = encountered_first_primitive
		set_policy = {
			policy = interference
			option = interference_subtle
		}
		set_policy_cooldown = interference
		set_policy = {
			policy = enlightenment
			option = enlightenment_not_allowed
		}
		set_policy_cooldown = enlightenment
		start_situation = {
			type = situation_mem_eden_protocol_protocolswill
		}
		random_situation = {
			limit = {
				is_situation_type = situation_mem_eden_protocol_protocolswill
			}
			set_situation_approach = approach_mem_eden_protocol_protocolswill_1
		}
		begin_event_chain = {
			event_chain = mem_eden_protocol_chain
		}
		capital_scope = {
			add_building = building_mem_eden_protocol_2
			# Otherwise we start in negative 
			add_district = district_hab_energy
			add_district = district_hab_energy
			while = {
				count = 2
				closest_system = {
					limit = {
						OR = {
							has_star_flag = neighbor_t1_first_colony
							has_star_flag = neighbor_t2_second_colony
						}
						NOT = {
							any_system_planet = {
								has_planet_flag = mem_eden_protocol_terraforming_candidate
							}
						}
					}
					random_system_planet = {
						limit = {
							has_planet_flag = prescripted_ideal
						}
						clear_deposits = yes
						change_pc = pc_barren
						add_modifier = {
							modifier = terraforming_candidate
						}
						set_planet_flag = mem_eden_protocol_terraforming_candidate
					}
					if = {
						limit = {
							NOT = {
								any_system_planet = {
									has_planet_flag = mem_eden_protocol_terraforming_candidate
								}
							}
						}
						random_system_planet = {
							limit = {
								is_star = no
								planet_size > 8
								planet_size < 26
							}
							clear_deposits = yes
							clear_planet_modifiers = yes
							change_pc = pc_barren
							add_modifier = {
								modifier = terraforming_candidate
							}
							set_planet_flag = mem_eden_protocol_terraforming_candidate
						}
					}
				}
			}
			solar_system = {
				random_system_planet = {
					limit = {
						exists = owner
						owner = {
							is_country_type = primitive
						}
					}
					owner = {
						establish_communications_no_message = root
					}
				}
			}
		}
	}
}

#yearly pulse event
country_event = {
	id = mem_eden_protocol.2
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		has_origin = origin_mem_eden_protocol
	}
	immediate = {
		if = {
			limit = {
				NOT = {
					has_valid_civic = civic_machine_assimilator
				}
			}
			set_policy_cooldown = interference
			set_policy_cooldown = enlightenment
		}
		if = {
			limit = {
				has_country_flag = found_presapients
				NOT = {
					has_country_flag = mem_eden_protocol_pre_sapient
				}
				NOT = {
					has_valid_civic = civic_machine_assimilator
				}
			}
			set_country_flag = mem_eden_protocol_pre_sapient
			set_policy = {
				policy = pre_sapients
				option = pre_sapients_protect
			}
		}
		every_owned_planet = {
			limit = {
				is_planet_class = pc_habitat
			}
			# Roll for seeding
			if = {
				limit = {
					owner = {
						NOT = {
							has_country_flag = mem_eden_protocol_seeding_pause
						}
					}
					OR = {
						has_building = building_mem_eden_protocol_2
						has_building = building_mem_eden_protocol_3
					}
					solar_system = {
						any_system_planet = {
							habitable_planet_not_urban = yes
							NOT = {
								exists = owner
							}
						}
					}
				}
				random_list = {
					80 = {
					}
					20 = {
						solar_system = {
							random_system_planet = {
								limit = {
									habitable_planet_not_urban = yes
									NOT = {
										exists = owner
									}
								}
								set_planet_flag = mem_eden_protocol_seeding_target
							}
						}
						owner = {
							set_timed_country_flag = {
								flag = mem_eden_protocol_seeding_pause
								years = 2
							}
							country_event = {
								id = mem_eden_protocol.3
								days = 1
								random = 358
							}
						}
					}
				}
				reroll_random = yes
			}
			# Freeze Pre-FTL progress
			if = {
				limit = {
					OR = {
						has_building = building_mem_eden_protocol_2
						has_building = building_mem_eden_protocol_3
					}
					solar_system = {
						any_system_planet = {
							exists = owner
							owner = {
								is_country_type = primitive
								OR = {
									has_country_flag = machine_age
									has_country_flag = atomic_age
									has_country_flag = early_space_age
								}
							}
						}
					}
				}
				solar_system = {
					every_system_planet = {
						limit = {
							exists = owner
							owner = {
								is_country_type = primitive
								OR = {
									has_country_flag = machine_age
									has_country_flag = atomic_age
									has_country_flag = early_space_age
								}
							}
						}
						owner = {
							set_country_flag = tech_frozen
						}
					}
				}
			}
			if = {
				limit = {
					NOT = {
						is_variable_set = mem_eden_protocol_terraforming_progress
					}
				}
				set_variable = {
					which = mem_eden_protocol_terraforming_progress
					value = 0
				}
			}
			# Roll for standard terraforming for terraforming candidates
			if = {
				limit = {
					owner = {
						NOT = {
							has_country_flag = mem_eden_protocol_terraforming_pause
						}
					}
					OR = {
						has_building = building_mem_eden_protocol_1
						has_building = building_mem_eden_protocol_2
						has_building = building_mem_eden_protocol_3
					}
					solar_system = {
						any_system_planet = {
							has_modifier = terraforming_candidate
							is_terraforming = no
						}
					}
				}
				random_list = {
					60 = {
					}
					1 = {
						modifier = {
							check_variable = {
								which = mem_eden_protocol_terraforming_progress
								value > 0
							}
							factor = mem_eden_protocol_terraforming_progress
						}
						modifier = {
							check_variable = {
								which = mem_eden_protocol_terraforming_progress
								value < 1
							}
							factor = 0
						}
						set_variable = {
							which = mem_eden_protocol_terraforming_progress
							value = -1
						}
						solar_system = {
							random_system_planet = {
								limit = {
									has_modifier = terraforming_candidate
									is_terraforming = no
								}
								set_planet_flag = mem_eden_protocol_terraforming_target
							}
						}
						owner = {
							set_timed_country_flag = {
								flag = mem_eden_protocol_terraforming_pause
								years = 3
							}
							country_event = {
								id = mem_eden_protocol.3
								days = 1
								random = 358
							}
						}
					}
				}
				reroll_random = yes
			}
			# Roll for standard terraforming for special terraforming candidates (for tier 3 building)
			if = {
				limit = {
					owner = {
						NOT = {
							has_country_flag = mem_eden_protocol_terraforming_pause
						}
					}
					has_building = building_mem_eden_protocol_3
					solar_system = {
						any_system_planet = {
							OR = {
								has_modifier = toxic_terraforming_candidate
								has_modifier = frozen_terraforming_candidate
							}
							is_terraforming = no
						}
					}
				}
				random_list = {
					60 = {
					}
					1 = {
						modifier = {
							check_variable = {
								which = mem_eden_protocol_terraforming_progress
								value > 0
							}
							factor = mem_eden_protocol_terraforming_progress
						}
						modifier = {
							check_variable = {
								which = mem_eden_protocol_terraforming_progress
								value < 1
							}
							factor = 0
						}
						set_variable = {
							which = mem_eden_protocol_terraforming_progress
							value = -1
						}
						solar_system = {
							random_system_planet = {
								limit = {
									OR = {
										has_modifier = toxic_terraforming_candidate
										has_modifier = frozen_terraforming_candidate
									}
									is_terraforming = no
								}
								set_planet_flag = mem_eden_protocol_terraforming_target
							}
						}
						owner = {
							set_timed_country_flag = {
								flag = mem_eden_protocol_terraforming_pause
								years = 3
							}
							country_event = {
								id = mem_eden_protocol.3
								days = 1
								random = 358
							}
						}
					}
				}
				reroll_random = yes
			}
			# Roll for gaia terraforming for habitable planets (for tier 3 building)
			if = {
				limit = {
					owner = {
						NOT = {
							has_country_flag = mem_eden_protocol_terraforming_pause
						}
					}
					has_building = building_mem_eden_protocol_3
					solar_system = {
						any_system_planet = {
							OR = {
								habitable_planet_not_urban = yes
								is_planet_class = pc_nuked
							}
							NOT = {
								is_planet_class = pc_gaia
							}
							OR = {
								NOT = {
									exists = owner
								}
								owner = {
									is_country_type = primitive
								}
							}
							is_terraforming = no
						}
					}
				}
				random_list = {
					80 = {
					}
					1 = {
						modifier = {
							check_variable = {
								which = mem_eden_protocol_terraforming_progress
								value > 0
							}
							factor = mem_eden_protocol_terraforming_progress
						}
						modifier = {
							check_variable = {
								which = mem_eden_protocol_terraforming_progress
								value < 1
							}
							factor = 0
						}
						set_variable = {
							which = mem_eden_protocol_terraforming_progress
							value = -1
						}
						solar_system = {
							random_system_planet = {
								limit = {
									OR = {
										habitable_planet_not_urban = yes
										is_planet_class = pc_nuked
									}
									NOT = {
										is_planet_class = pc_gaia
									}
									OR = {
										NOT = {
											exists = owner
										}
										owner = {
											is_country_type = primitive
										}
									}
									is_terraforming = no
								}
								set_planet_flag = mem_eden_protocol_terraforming_target_gaia
							}
						}
						owner = {
							set_timed_country_flag = {
								flag = mem_eden_protocol_terraforming_pause
								years = 3
							}
							country_event = {
								id = mem_eden_protocol.3
								days = 1
								random = 358
							}
						}
					}
				}
				reroll_random = yes
			}
			# increase terraforming variable by 1 ( if terraforming has happened, the outcome will be 0 ), if there is a valid planet for terraforming
			if = {
				limit = {
					OR = {
						has_building = building_mem_eden_protocol_1
						has_building = building_mem_eden_protocol_2
						has_building = building_mem_eden_protocol_3
					}
					solar_system = {
						any_system_planet = {
							OR = {
								NOT = {
									exists = owner
								}
								AND = {
									exists = owner
									owner = {
										is_country_type = primitive
									}
								}
							}
							OR = {
								AND = {
									# back to check on habitat for this condition
									prevprev = {
										has_building = building_mem_eden_protocol_3
									}
									OR = {
										has_modifier = frozen_terraforming_candidate
										has_modifier = toxic_terraforming_candidate
										AND = {
											habitable_planet_not_urban = yes
											NOT = {
												is_planet_class = pc_gaia
											}
										}
										is_planet_class = pc_nuked
									}
								}
								has_modifier = terraforming_candidate
							}
						}
					}
				}
				change_variable = {
					which = mem_eden_protocol_terraforming_progress
					value = 1
				}
			}
			# if there is no valid target progress is reset to 0
			else = {
				set_variable = {
					which = mem_eden_protocol_terraforming_progress
					value = 0
				}
			}
			save_event_target_as = mem_eden_protocol_habitat
		}
	}
}

# Terraforming or Seeding starts
country_event = {
	id = mem_eden_protocol.3
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		# normal terraforming
		if = {
			limit = {
				any_planet_within_border = {
					has_planet_flag = mem_eden_protocol_terraforming_target
				}
			}
			random_planet_within_border = {
				limit = {
					has_planet_flag = mem_eden_protocol_terraforming_target
				}
				remove_planet_flag = mem_eden_protocol_terraforming_target
				# in case someone started terraforming this planet in the meantime, atleast remove the blocking flag
				if = {
					limit = {
						is_terraforming = yes
					}
					root = {
						remove_country_flag = mem_eden_protocol_terraforming_pause
					}
				}
				else = {
					start_terraform_process = rl_habitable_normal
				}
			}
		}
		# gaia terraforming
		if = {
			limit = {
				any_planet_within_border = {
					has_planet_flag = mem_eden_protocol_terraforming_target_gaia
				}
			}
			random_planet_within_border = {
				limit = {
					has_planet_flag = mem_eden_protocol_terraforming_target_gaia
				}
				remove_planet_flag = mem_eden_protocol_terraforming_target_gaia
				# in case someone started terraforming this planet in the meantime, atleast remove the blocking flag
				if = {
					limit = {
						is_terraforming = yes
					}
					root = {
						remove_country_flag = mem_eden_protocol_terraforming_pause
					}
				}
				else = {
					start_terraform_process = pc_gaia
				}
			}
		}
		# Seeding
		if = {
			limit = {
				any_planet_within_border = {
					has_planet_flag = mem_eden_protocol_seeding_target
				}
			}
			random_planet_within_border = {
				limit = {
					has_planet_flag = mem_eden_protocol_seeding_target
				}
				start_colony = {
					owner = root
					species = root
				}
				remove_planet_flag = mem_eden_protocol_seeding_target
			}
		}
	}
}

# A planet has been colonized.
# Scope = Planet
# give it to pre-ftl
planet_event = {
	id = mem_eden_protocol.6
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		exists = space_owner
		space_owner = {
			has_origin = origin_mem_eden_protocol
			NOT = {
				has_valid_civic = civic_machine_assimilator
			}
		}
		mem_eden_pre_ftl_possible_planet = yes
	}
	immediate = {
		save_event_target_as = mem_eden_protocol_seeding_target
		owner = {
			set_variable = {
				which = mem_eden_protocol_seeding_target
				value = event_target:mem_eden_protocol_seeding_target.trigger:planet_size
			}
			multiply_variable = {
				which = mem_eden_protocol_seeding_target
				value = 5
			}
			mem_eden_protocol_add_protocol_progress = {
				AMOUNT = owner.mem_eden_protocol_seeding_target
			}
			
			if = {
				limit = {
					any_situation = {
						is_situation_type = situation_mem_eden_protocol_protocolswill
					}
				}
				create_message = {
					type = MESSAGE_MEM_MADE_PRIMITIVE
					target = root
					localization = MESSAGE_MEM_MADE_PRIMITIVE_DESC
					variable = {
						type = variable
						localization = PROGRESS
						varname = mem_eden_protocol_seeding_target
						scope = this
					}
					variable = {
						type = name
						localization = PLANET
						scope = root 
					}
					days = 30
				}
