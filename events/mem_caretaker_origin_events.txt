planet_event = {
	id = mem_caretaker_origin_event.10
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		mem_caretaker_presence_on_planet = yes
		FROM = {
			NOT = {
				has_origin = origin_mem_caretaker
			}
		}
	}
	immediate = {
		mem_caretaker_hub_clean_up = yes
		remove_modifier = mem_caretaker_hacked_hub_1
		remove_modifier = mem_caretaker_hacked_hub_2
		remove_modifier = mem_caretaker_hacked_hub_3a
		remove_modifier = mem_caretaker_hacked_hub_3b
		remove_deposit = d_mem_caretaker_free_settlements
		remove_modifier = pm_mem_caretaker_free_settler_discord
	}
}

### empire with mem_caretaker origin takes over a colony of someone else : schedule mem_caretaker hub construction events
planet_event = {
	id = mem_caretaker_origin_event.20
	is_triggered_only = yes
	hide_window = yes
	trigger = {
		FROM = {
			has_origin = origin_mem_caretaker
		}
	}
	immediate = {
		# country has active mem_caretaker
		if = {
			limit = {
				FROM = {
					mem_caretaker_tech_access = yes
				}
			}
			# planet doesn't have mem_caretaker core -> schedule event for checking if the core was built
			if = {
				limit = {
					NOT = {
						has_building = building_mem_caretaker_outreach
					}
				}
				planet_event = {
					id = mem_caretaker_origin_event.1110
					days = 360
				}

###  - if the colonization process hasn't finished yet, retry check later (on_action triggers appear to be borked as discovered by Ash)
###  - if colonization is finished, call actual checking event in 360 days
planet_event = {
	id = mem_caretaker_origin_event.1110
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		exists = owner
		owner = {
			has_origin = origin_mem_caretaker
			NOT = {
				has_country_flag = mem_caretaker_break_up
			}
		}
		mem_caretaker_presence_on_planet = no
		NOT = {
			has_planet_flag = mem_caretaker_origin_reminded_to_build_mem_caretaker_hub
		}
	}
	immediate = {
		if = {
			limit = {
				is_colony = yes
				pop_amount > 0
				NOT = {
					owner = {
						has_policy_flag = mem_caretaker_policy_autobuild
					}
				}
			}
			log = "Colony has been established; Remind to build a Caretaker hub in 360 days."
			planet_event = {
				id = mem_caretaker_origin_event.1111
				days = 360
			}
		}
		else_if = {
			limit = {
				is_colony = yes
				pop_amount > 0
				owner = {
					has_policy_flag = mem_caretaker_policy_autobuild
				}
			}
			add_building = building_mem_caretaker_outreach
		}
		else = {
			log = "Colonization is still in progress, try later..."
			planet_event = {
				id = mem_caretaker_origin_event.1110
				days = 180
			}
		}
	}
}
