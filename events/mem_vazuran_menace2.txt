# Setup Event at Game Start for randomly disabling the Crisis
event = {
	id = mem_vazurans.1
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		IF = {
			limit = {
				any_country = {
					has_origin = origin_mem_fainthearted
				}
			}
			set_global_flag = mem_vazurans_disabled
			# Nothing happens. Fainthearted Backwater wins over From the Ashes.
		}
		ELSE_IF = {
			limit = {
				any_country = {
					has_origin = origin_mem_from_the_ashes
				}
			}
			start_mem_vazuran_menace = yes
		}
		ELSE = {
			random_list = {
				95 = {
					set_global_flag = mem_vazurans_disabled
				}
				5 = {
					start_mem_vazuran_menace = yes
				}
			}
		}
		IF = {
			limit = {
				has_global_flag = mem_vazurans_disabled
			}
			random_list = {
				80 = {
				}
				# nothing happens
				20 = {
					setup_mem_ruined_vazuran_space = yes
				}
			}
		}
		mem_gamestart_orila_primitive_setup = yes
	}
}

country_event = {
	id = mem_vazurans.1120
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		#create_mem_vazuran_country = yes wont need this hopefully
		from = {
			if = {
				limit = {
					any_system_planet = {
						has_planet_flag = mem_vazurans_raiding_target
					}
				}
				random_system_ambient_object = {
					limit = {
						has_ambient_object_flag = mem_vazurans_rift
					}
					create_mem_vazuran_fleet = yes
					#mem_vazuran_raiding_fleet_spawn = yes
				}
			}
		}
		last_created_fleet = {
			# fleet flag for bombarding event
			set_fleet_flag = mem_vazurans_fleet
			set_fleet_stance = aggressive
			set_aggro_range_measure_from = self
			set_aggro_range = 150
			fleet_event = {
				id = mem_vazurans.110
				days = 1
			}
		}
		event_target:global_event_country = {
			change_variable = {
				which = mem_vazuran_menace_raids_variable
				value = 1
			}
		}
	}
}
