# Setup Event at Game Start for randomly disabling the Crisis
event = {
	id = mem_vazurans.1
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		IF = {
			limit = {
				any_country = {
					has_origin = origin_mem_fainthearted
				}
			}
			set_global_flag = mem_vazurans_disabled
			# Nothing happens. Fainthearted Backwater wins over From the Ashes.
		}
		ELSE_IF = {
			limit = {
				any_country = {
					has_origin = origin_mem_from_the_ashes
				}
			}
			start_mem_vazuran_menace = yes
		}
		ELSE = {
			random_list = {
				95 = {
					set_global_flag = mem_vazurans_disabled
				}
				5 = {
					start_mem_vazuran_menace = yes
				}
			}
		}
		IF = {
			limit = {
				has_global_flag = mem_vazurans_disabled
			}
			random_list = {
				80 = {
				}
				# nothing happens
				20 = {
					setup_mem_ruined_vazuran_space = yes
				}
			}
		}
		mem_gamestart_orila_primitive_setup = yes
	}
}

# vazuran ground combat won 
# Triggers country_event for the attacker upon victory (before controller is switched)
# This = country, lead attacker
# From = country, planet owner
# FromFrom = planet
country_event = {
	id = mem_vazurans.1131
	hide_window = yes
	is_triggered_only = yes
	trigger = {
		FromFrom = {
			has_planet_flag = mem_vazurans_raid_planet
		}
	}
	immediate = {
		FromFrom = {
			# armies done their job 
			remove_planet_flag = mem_vazurans_raid_planet
			every_planet_army = {
				limit = {
					owner = {
						is_country_type = mem_vazurans
					}
				}
				remove_army = yes
			}			
			#set_controller = from # TRODO this needs testing, set owner back actual owner
			# return to rift
			solar_system = {
				every_fleet_in_system = {
					limit = {
						owner = {
							is_country_type = mem_vazurans
						}
					}
					fleet_event = {
						id = mem_vazurans.1135
						days = 1
					}
				}
			}

			IF = {
				limit = {
					OR = {
						AND = {
							has_global_flag = galactic_focus_crisis_vazurans_1
							owner = {
								is_galactic_community_member = yes
							}
						}
						check_modifier_value = {
							modifier = planet_orbital_bombardment_damage
							value <= 0.5
						}
					}
				}
				add_planet_devastation = 40
				while = {
					count = value:perc_of_planet_pop					# 30% maybe too much easily changed
					# this makes it only  around 15%
					random_list = {
						50 = {
							random_owned_pop = {
								kill_pop = yes
							}
						}
						50 = {
						}
					}
				}
				if = {
					limit = {
						habitable_structure = yes
					}
					while = {
						count = 3
						add_deposit = d_mem_vazuran_ruined_section
					}
				}
				else = {
					while = {
						count = 2
						add_deposit = d_mem_vazuran_bomb_crater
					}
					while = {
						count = 1
						add_deposit = d_city_ruins
					}
				}
			}
			else = {
				add_planet_devastation = 100
				while = {
					count = value:perc_of_planet_pop					# 30% maybe too much easily changed
					random_owned_pop = {
						kill_pop = yes
					}
				}
				if = {
					limit = {
						habitable_structure = yes
					}
					while = {
						count = 5
						add_deposit = d_mem_vazuran_ruined_section
					}
				}
				else = {
					while = {
						count = 2
						add_deposit = d_mem_vazuran_bomb_crater
					}
					while = {
						count = 2
						add_deposit = d_city_ruins
					}
				}
			}
			
		}
	}
}
