country_event = {
	id = mem_scfe_ziaskehorn.31	# 02
	title = ziaskehorn.02.name
	picture = GFX_evt_archaeology_trap
	desc = ziaskehorn.02.desc
	show_sound = event_bad_omen
	is_triggered_only = yes
	option = {
		name = UNFORTUNATE
		custom_tooltip = ziaskehorn.02.tooltip
	}
	after = {
		hidden_effect = {
			add_modifier = {
				modifier = mem_scfe_embers_of_ziaskehorn
				days = 360
			}
			every_owned_leader = {
				limit = {
					leader_class = commander
				}
				# Traits cleanup
				if = {
					limit = {
						has_trait = leader_trait_restrained
					}
					remove_trait = leader_trait_restrained
				}
				if = {
					limit = {
						has_trait = leader_trait_nervous
					}
					remove_trait = leader_trait_nervous
				}
				else_if = {
					limit = {
						has_trait = leader_trait_nervous_2
					}
					remove_trait = leader_trait_nervous_2
				}
				if = {
					limit = {
						has_trait = leader_trait_lethargic
					}
					remove_trait = leader_trait_lethargic
				}
				else_if = {
					limit = {
						has_trait = leader_trait_lethargic_2
					}
					remove_trait = leader_trait_lethargic_2
				}
				# Adding Victory or Death
				if = {
					limit = {
						has_trait = leader_trait_victory_or_death
					}
					remove_trait = leader_trait_victory_or_death
					add_trait = {
						trait = leader_trait_victory_or_death_2
						show_message = no
						consume_selection = no
					}
				}
				else_if = {
					limit = {
						NOT = {
							has_trait = leader_trait_victory_or_death_2
						}
					}
					add_trait = {
						trait = leader_trait_victory_or_death
						show_message = no
						consume_selection = no
					}
				}
				# Adding Reckless
				if = {
					limit = {
						has_trait = leader_trait_reckless
					}
					remove_trait = leader_trait_reckless
					add_trait = {
						trait = leader_trait_reckless_2
						show_message = no
						consume_selection = no
					}
				}
				else_if = {
					limit = {
						NOT = {
							has_trait = leader_trait_reckless_2
						}
					}
					add_trait = {
						trait = leader_trait_reckless
						show_message = no
						consume_selection = no
					}
				}
			}
		}
	}
}
