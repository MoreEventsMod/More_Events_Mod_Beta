# Beginnings of Madness
fleet_event = {
	id = mem_kay_site_a.6
	title = mem_kay_site_a.6.name
	desc = mem_kay_site_a.6.desc
	picture = GFX_evt_mem_placeholder
	show_sound = event_default
	archaeology = yes
	is_triggered_only = yes
	trigger = {
		exists = leader
		leader = {
			NOT = {
				has_leader_flag = mem_kay_site_a_normalcy
			}
		}
	}
	option = {
		name = mem_kay_site_a.6.a
		leader = {
			unassign_leader = this
			add_trait = {
				trait = leader_trait_traumatized
			}
		}
	}
	option = {
		name = mem_kay_site_a.6.b
		leader = {
			unassign_leader = this
			add_trait = {
				trait = leader_trait_substance_abuser
			}
		}
	}
	option = {
		name = mem_kay_site_a.6.c
		leader = {
			random_list = {
				# Return to Normalcy
				50 = {
					leader_event = {
						id = mem_kay_site_a.7
						days = 30
					}
				}
				# Feral Scientist
				50 = {
					leader_event = {
						id = mem_kay_site_a.8
						days = 30
					}
				}
			}
		}
	}
}
