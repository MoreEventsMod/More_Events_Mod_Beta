# sending raiding fleet home to the rift and let them vanish
fleet_event = {
	id = mem_vazurans.1135
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		solar_system = {
			every_system_colony = {
				limit = {
					controller = {
						is_same_value = root.owner
					}
				}
				root.owner = {
					random_situation = {
						limit = {
							is_situation_type = mem_vazuran_raiding
						}
						# for each pop we get points
						change_variable = {
							which = mem_vazuran_raiding_strength
							value = prevprev.pop_amount
						}
						# for each mining/research staion we get points
						change_variable = {
							which = mem_vazuran_raiding_strength_temp
							value = prevprevprev.value:num_orbital_sites
						}
						multiply_variable = {
							which = mem_vazuran_raiding_strength_temp
							value = @mem_vazurans_points_per_mining_stations
						}
						change_variable = {
							which = mem_vazuran_raiding_strength
							value = mem_vazuran_raiding_strength_temp
						}
						clear_variable = mem_vazuran_raiding_strength_temp
					}
				}
				every_planet_army = {
					limit = {
						owner = {
							is_country_type = mem_vazurans
						}
					}
					remove_army = yes
				}
				set_controller = space_owner
			}
			root.owner = {
				random_situation = {
					limit = {
						is_situation_type = mem_vazuran_raiding
					}
					# let the fleet retreat
					change_variable = {
						which = mem_vazuran_raiding_current_fleets
						value = -1
					}
				}
			}
			random_system_ambient_object = {
				limit = {
					has_ambient_object_flag = mem_vazurans_rift
				}
				save_event_target_as = mem_vazurans_rift
			}
			every_fleet_in_system = {
				limit = {
					owner = {
						is_country_type = mem_vazurans
					}
				}
				fleet_event = {
					id = mem_vazurans.1140					# send all fleets back, including armies
				}
			}
		}
	}
}

# now lets decide wat outcome we had
situation_event = {
	id = mem_vazurans.1200
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		remove_global_flag = mem_vazuran_raiding
		############################## EMPTY VAZURAN HOMESYSTEM ########################################
		no_scope = {
			# makes system positions originate from galactic core
			spawn_system = {
				min_distance >= 550
				max_distance <= 560
				min_orientation_angle = 96
				max_orientation_angle = 102
				initializer = "mem_vazuran_homesystem"
				hyperlane = no
				# is_discovered = yes
			}
		}
		last_created_system = {
			spawn_megastructure = {
				type = mem_vazuran_gateway_nexus
				orbit_distance = 300
				orbit_angle = 360				# range of the randomization
			}
			random_system_megastructure = {
				limit = {
					is_megastructure_type = mem_vazuran_gateway_nexus
				}
				save_event_target_as = mem_vazuran_nexus_gate
				create_ambient_object = {
					type = "mem_vazuran_wormhole_big_object"
				}
				last_created_ambient_object = {
					set_location = {
						target = event_target:mem_vazuran_nexus_gate
						distance = 0
						angle = random
					}
				}
			}
			mem_vazurans_create_home_system_defences = yes
		}
		############################## FALLEN VAZURANS ########################################
		if = {
			limit = {
				check_variable_arithmetic = {
					which = mem_vazuran_raiding_strength
					value <= @mem_vazurans_ruined_state_threshold
				}
			}
			last_created_system = {
				random_system_planet = {
					limit = {
						is_planet_class = pc_mem_vazuran_habitat
					}
					# TRODO buildings and shit
				}
			}
			every_playable_country = {
				limit = {
					is_hostile_to = prev
				}
				country_event = {
					id = mem_vazurans.1101
				}
			}
		}
		############################## FALLEN EMPIRE VAZURANS ########################################
		else_if = {
			limit = {
				check_variable_arithmetic = {
					which = mem_vazuran_raiding_strength
					value <= @mem_vazurans_fallen_state_threshold
				}
			}
			event_target:mem_vazurans_country = {
				every_playable_country = {
					limit = {
						is_hostile_to = prev
					}
					set_faction_hostility = {
						set_neutral = yes
						set_friendly = no
						set_hostile = no
					}
				}
				set_country_type = fallen_empire
				capital_scope = {
					solar_system = {
						mem_vazurans_create_home_system_defences = yes
					}
				}
				# TRODO : portal to homesystem + setup scripted effect
				# notify players of outcome
			}
		}
		############################## AWAKENED EMPIRE VAZURANS ########################################
		else = {
			event_target:mem_vazurans_country = {
				every_playable_country = {
					limit = {
						is_hostile_to = prev
					}
					set_faction_hostility = {
						set_neutral = yes
						set_friendly = no
						set_hostile = no
					}
				}
				set_country_type = awakened_fallen_empire
				capital_scope = {
					solar_system = {
						mem_vazurans_create_home_system_defences = yes
					}
				}
				# TRODO : portal to homesystem + setup scripted effect
				# claim a lot of systems
				# notify players of outcome
			}
		}
	}
	after = {
		destroy_situation = this
	}
}
