# Setup Event at Game Start for randomly disabling the Crisis
event = {
	id = mem_vazurans.1
	hide_window = yes
	is_triggered_only = yes
	immediate = {
		if = {
			limit = {
				any_system = {
					has_star_flag = precursor_mem_vazurans
				}
			}
			log = "mem_vazurans:chosen"
			set_global_flag = mem_vazurans_chosen
		}
		# Options: someone chose ancestors and / or toggle precursor
		# Case 1: Ancestors off, Precursor on 	-> randomlist for active / ruined
		# Case 2: Ancestors on, Precursor on 	-> garantueed active
		# Case 3: Ancestors on, Precursor off 	-> advanced empire, not active
		# Case 4: Ancestors off, Precursor off 	-> nothing completly disabled anything related to vazurans
		IF = {
			limit = {
				OR = {
					any_playable_country = {
						is_mem_runesmith = yes
					}
					#has_global_flag = mem_vazurans_force_enable
				}
			}
			if = {
				limit = {
					has_global_flag = mem_vazurans_chosen
				}
				event_target:global_event_country = {
					country_event = {
						id = mem_vazurans.2
						days = @mem_vazurans_starting_date
						random = @mem_vazurans_starting_date_random
					}
				}
				no_scope = {
					create_mem_vazuran_destroyed_system_chain = yes
				}
				log = "mem_vazurans:active"
			}
			else = {
				log = "mem_vazurans:advanced"
				event_target:global_event_country = {
					country_event = {
						id = mem_ancestors_grudge.2
					}
				}
				# this covers the case when people want to spawn them but forget to precursor settings. in which case we have the advanced version
				# if = {
				# 	limit = {
				# 		has_global_flag = mem_vazurans_force_enable
				# 	}
				# 	event_target:global_event_country = {
				# 		country_event = {
				# 			id = mem_vazurans.2
				# 			days = @mem_vazurans_starting_date
				# 			random = @mem_vazurans_starting_date_random
				# 		}
				# 	}
				# 	no_scope = {
				# 		create_mem_vazuran_destroyed_system_chain = yes
				# 	}
				# 	log = "mem_vazurans:active"
				# }
			}
		}
		else = {
			if = {
				limit = {
					has_global_flag = mem_vazurans_chosen
				}
				log = "mem_vazurans:random"
				random_list = {
					90 = {
						reroll_random = yes
						random_list = {
							80 = {
							}
							# nothing happens
							20 = {
								setup_mem_ruined_vazuran_space = yes
							}
						}
					}
					10 = {
						event_target:global_event_country = {
							country_event = {
								id = mem_vazurans.2
								days = @mem_vazurans_starting_date
								random = @mem_vazurans_starting_date_random
							}
						}
						no_scope = {
							create_mem_vazuran_destroyed_system_chain = yes
						}
					}
				}
			}
			else = {
				# nothing happens player choose neither ancestors nor precursor setting		
				log = "mem_vazurans:inactive"
			}
		}
		mem_gamestart_orila_primitive_setup = yes
	}
}
