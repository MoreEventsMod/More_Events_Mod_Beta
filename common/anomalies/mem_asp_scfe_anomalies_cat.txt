mem_ASP_epilogue_anomaly = {
	desc = mem_ASP_epilogue_anomaly_desc
	picture = GFX_evt_debris
	level = 4
	
	max_once_global = yes
	should_ai_and_humans_use = yes

	spawn_chance = {
		modifier = {
			add = 10
			NOR = {
				is_star = yes
				is_asteroid = yes
			}
			solar_system = {
				exists = star
				NOR = {
					exists = archaeological_site
					any_system_planet = { has_anomaly = yes } #there should be no anomaly in the system
					has_star_flag = empire_cluster
					is_fe_cluster = yes
				}
			}
			# from.owner = {
			# 	is_nomad = no
			# }
			from.owner.capital_scope.solar_system = {
				NOR = {
					exists = archaeological_site
					any_system_planet = { has_anomaly = yes } #there should be no anomaly in the system
					
				}
			}
		}
	}

	on_spawn = {
		create_ambient_object = {
			type = "large_debris_object" #행성의 궤도에 무수한 쓰레기들이 부유하고 있습니다.
			entity_offset_height = -15 #just right fit to target
			location = root
		}
		last_created_ambient_object = {
			set_location = {
				target = root
				distance = 0
				angle = random
			}
		}
	}

	on_success = mem_asp.78
}
