	time = 360
	is_for_colonizable = yes
	category = deposit_cat_blockers

		always = no
	}

		planet_max_districts_add = -2
	}
	resources = {
		category = deposit_blockers
		cost = {
			energy = 1200
		}
	}
	on_cleared = {
		#custom_tooltip = mem_ancestors_grudge_boosts_repairs_effect
		if = {
			limit = {
				host_has_dlc = "Ancient Relics Story Pack"
				exists = owner
			}
			owner = {
				add_resource = {
					minor_artifacts = 75
				}
			}
		}
		hidden_effect = {
			if = {
				limit = {
					has_planet_flag = mem_ancestors_grudge_dranirs_hall_khazad_dum
				}
				planet_event = {
					id = mem_ancestors_grudge_dranir.30
				}
			}
			else = {
				owner = {
					country_event = {
						id = mem_ancestors_grudge.30
					}
				}
			}
		}
	}
	drop_weight = {
		weight = 0
	}
}
