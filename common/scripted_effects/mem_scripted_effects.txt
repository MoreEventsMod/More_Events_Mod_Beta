mem_add_proper_species_trait = {
	if = {
		limit = {
			is_species_class = LITHOID
		}
		change_species_characteristics = {
			add_trait = trait_lithoid
		}
	}
	else_if = {
		limit = {
			is_species_class = INF
		}
		change_species_characteristics = {
			add_trait = trait_infernal
		}
	}
	else = {
		change_species_characteristics = {
			add_trait = trait_organic
		}
	}
}

mem_start_event_purging = {
	create_pop_group = {
		species = owner.species
		category = purge
		effect = {
			save_event_target_as = purge_pop_group
			set_pop_group_flag = event_purge
		}
	}
	every_owned_pop_group = {
		limit = {
			NOT = {
				is_same_value = event_target:purge_pop_group
			}
		}
		owner = {
			add_resource = {
				entropy_crystals = $AMOUNT$
			}
		}
	}
}
