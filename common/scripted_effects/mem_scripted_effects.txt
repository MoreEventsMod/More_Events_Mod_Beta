mem_add_proper_species_trait = {
	if = {
		limit = {
			is_species_class = LITHOID
		}
		change_species_characteristics = {
			add_trait = trait_lithoid
		}
	}
	else_if = {
		limit = {
			is_species_class = INF
		}
		change_species_characteristics = {
			add_trait = trait_infernal
		}
	}
	else = {
		change_species_characteristics = {
			add_trait = trait_organic
		}
	}
}
