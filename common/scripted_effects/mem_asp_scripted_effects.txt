mem_aspmod_geo_centralizer_effect = {
	save_event_target_as = swaped_target_world
	if = {
		limit = {
			NOR = {
				is_planet_class = pc_habitat
				is_planet_class = pc_ringworld_habitable
			}
		}
		solar_system = {
			if = {
				limit = {
					NOR = {
						has_star_flag = already_swaped_system
						star = {
							is_planet_class = pc_black_hole
						}
					}
				}
				if = {
					limit = {
						OR = {
							has_planet_class = pc_b_star
							has_planet_class = pc_a_star
							has_planet_class = pc_f_star
							has_planet_class = pc_g_star
							has_planet_class = pc_k_star
							has_planet_class = pc_m_star
							has_planet_class = pc_t_star
							has_planet_class = pc_pulsar
							has_planet_class = pc_m_giant_star
							has_planet_class = pc_neutron_star
						}
					}
					switch = {
						trigger = has_planet_class
						pc_b_star = {
							spawn_planet = {
								class = pc_b_star
								location = prev
								init_effect = {
									set_name = star
									set_surveyed = {
										surveyed = yes
										surveyor = from.owner
									}
								}
							}
						}
						pc_a_star = {
							spawn_planet = {
								class = pc_a_star
								location = prev
								init_effect = {
									set_name = star
									set_surveyed = {
										surveyed = yes
										surveyor = from.owner
									}
								}
							}
						}

							spawn_planet = {
								class = pc_f_star
								location = prev
								init_effect = {
									set_name = star
									set_surveyed = {
										surveyed = yes
										surveyor = from.owner
									}
								}
							}
						}
						pc_g_star = {
							spawn_planet = {
								class = pc_g_star
								location = prev
								init_effect = {
									set_name = star
									set_surveyed = {
										surveyed = yes
										surveyor = from.owner
									}
								}
							}
						}
						pc_k_star = {
							spawn_planet = {
								class = pc_k_star
								location = prev
								init_effect = {
									set_name = star
									set_surveyed = {
										surveyed = yes
										surveyor = from.owner
									}
								}
							}
						}
						pc_m_star = {
							spawn_planet = {
								class = pc_m_star
								location = prev
								init_effect = {
									set_name = star
									set_surveyed = {
										surveyed = yes
										surveyor = from.owner
									}
								}
							}
						}
						pc_t_star = {
							spawn_planet = {
								class = pc_t_star
								location = prev
								init_effect = {
									set_name = star
									set_surveyed = {
										surveyed = yes
										surveyor = from.owner
									}
								}
							}
						}
						pc_pulsar = {
							spawn_planet = {
								class = pc_pulsar

								location = prev

							}
						}
						pc_neutron_star = {

								class = pc_neutron_star
								location = prev
								init_effect = {
									set_name = star
									set_surveyed = {
										surveyed = yes
										surveyor = from.owner
									}
								}
							}
						}
					}
					else = {
						spawn_planet = {
							class = pc_pulsar
							location = prev
							init_effect = {
								set_planet_size = 25
								set_name = star
								set_surveyed = {
									surveyed = yes
									surveyor = from.owner
								}
							}
						}
					}
				}
				spawn_planet = {
					class = pc_barren_cold
					size = 25
					orbit_distance = 0.01					#0 causes a graphical glitch #stellaris v2.8.1
					has_ring = no
					init_effect = {
						set_name = event_target:swaped_target_world
						clear_planet_modifiers = yes
						save_event_target_as = already_swaped_world
						random_list = {
							1 = {
								add_deposit = d_minerals_3
								add_deposit = d_minerals_3
								add_deposit = d_minerals_3
								add_deposit = d_minerals_3
								add_deposit = d_minerals_3

								add_deposit = d_minerals_3
								add_deposit = d_minerals_3

								add_deposit = d_minerals_3
							}

								add_deposit = d_minerals_3
								add_deposit = d_minerals_3
								add_deposit = d_minerals_3
							}
							1 = {
								add_deposit = d_minerals_3
								add_deposit = d_minerals_3
							}
						}
						set_surveyed = {
							surveyed = yes
							surveyor = from.owner
						}
					}
				}
				star = {
					clear_deposits = yes
					set_planet_size = 0					#hide star
				}
				set_star_flag = already_swaped_system				#shouldn't swap star with a planet more than once
				prev = {
					remove_planet = yes
					from = {
						set_location = {
							target = event_target:already_swaped_world
							distance = 0
						}
					}
				}
			}
			if = {
				limit = {
					OR = {
						has_star_flag = already_swaped_system
						star = {
							is_planet_class = pc_black_hole
						}
					}
				}
				prev = {
					kill_all_pop = yes
					remove_all_buildings = yes
					change_pc = pc_barren_cold
				}
			}
		}
	}
	if = {
		limit = {
			is_planet_class = pc_habitat
		}
		spawn_habitat_cracker_effect = yes
		remove_planet = yes
	}
	if = {
		limit = {
			is_planet_class = pc_ringworld_habitable
		}
		spawn_ringworld_cracker_effect = yes
		change_pc = pc_ringworld_habitable_damaged
	}
}

mem_asp_black_marker_dead_armies_count = {
	switch = {
		trigger = has_modifier
		mem_aspmod_deadspace_marker_relic_necromorph_slayer = {
			remove_modifier = mem_aspmod_deadspace_marker_relic_necromorph_slayer
		}
		mem_aspmod_deadspace_marker_relic_necromorph_slayer_necromancer = {
			remove_modifier = mem_aspmod_deadspace_marker_relic_necromorph_slayer_necromancer
		}
	}
	change_variable = {
		which = mem_asp_black_marker_dead_armies_count
		value = 1
	}
	if = {
		limit = {
			mem_asp_is_necro_dude = yes
		}
		add_modifier = {
			modifier = mem_aspmod_deadspace_marker_relic_necromorph_slayer_necromancer
			mult = mem_asp_black_marker_dead_armies_count
			days = 1800
		}
	}
	else = {
		add_modifier = {
			modifier = mem_aspmod_deadspace_marker_relic_necromorph_slayer
			mult = mem_asp_black_marker_dead_armies_count
			days = 1800
		}
	}
}
