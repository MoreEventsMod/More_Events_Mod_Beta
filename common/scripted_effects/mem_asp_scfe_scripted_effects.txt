mem_asp_black_marker_dead_armies_count = {
	switch = {
		trigger = has_modifier
		mem_aspmod_deadspace_marker_relic_necromorph_slayer = {
			remove_modifier = mem_aspmod_deadspace_marker_relic_necromorph_slayer
		}
		mem_aspmod_deadspace_marker_relic_necromorph_slayer_necromancer = {
			remove_modifier = mem_aspmod_deadspace_marker_relic_necromorph_slayer_necromancer
		}
	}
	change_variable = {
		which = mem_asp_black_marker_dead_armies_count
		value = 1
	}
	if = {
		limit = {
			mem_asp_is_necro_dude = yes
		}
		add_modifier = {
			modifier = mem_aspmod_deadspace_marker_relic_necromorph_slayer_necromancer
			mult = mem_asp_black_marker_dead_armies_count
			days = 1800
		}
	}
	else = {
		add_modifier = {
			modifier = mem_aspmod_deadspace_marker_relic_necromorph_slayer
			mult = mem_asp_black_marker_dead_armies_count
			days = 1800
		}
	}
}
