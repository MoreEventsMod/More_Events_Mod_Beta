#sets up the systems around empires and activates the crisis
start_mem_vazuran_menace = {
	if = {
		limit = {
			NOT = {
				has_global_flag = mem_vazurans_spawned
			}
		}
		#set_global_flag = mem_vazurans_enabled
		create_mem_vazuran_country = yes
		set_global_flag = mem_vazurans_spawned
		event_target:mem_vazurans_country = {
			log = this.id
			# situation controls all raiding
			start_situation = {
				type = mem_vazuran_raiding
			}
			random_situation = {
				limit = {
					is_situation_type = mem_vazuran_raiding
				}
				# max number of fleets that can raid simultaniously
				set_variable = {
					which = mem_vazuran_raiding_max_fleets
					value = value:mem_vazuran_raids_per_country
				}
				# max concurrent number of fleets raiding, could be done in one variable but this is more stable
				set_variable = {
					which = mem_vazuran_raiding_current_fleets
					value = 0
				}
				# tracks all, raided resources, lost fleets etc. should be deciding factor for outcome
				set_variable = {
					which = mem_vazuran_raiding_strength					# make this impactful
					value = @mem_vazurans_starting_points
				}
			}
		}
	}
}

# create the destroyed worlds systems
create_mem_vazuran_destroyed_system_chain = {
	every_country = {
		limit = {
			is_country_type = default
		}
		random_list = {
			1 = {
				capital_scope = {
					solar_system = {
						random_neighbor_system = {
							random_neighbor_system = {
								limit = {
									NOT = {
										has_star_flag = empire_home_system
									}
									any_system_planet = {
										OR = {
											is_planet_class = pc_barren
											is_planet_class = pc_barren_cold
											is_planet_class = pc_toxic
											is_planet_class = pc_frozen
										}
									}
								}
								set_star_flag = mem_vazuran_destroyed_planet_system
								random_system_planet = {
									limit = {
										OR = {
											is_planet_class = pc_barren
											is_planet_class = pc_barren_cold
											is_planet_class = pc_toxic
											is_planet_class = pc_frozen
										}
									}
									random_list = {
										1 = {
											change_pc = pc_shattered
										}
										1 = {
											change_pc = pc_broken
										}
									}
									set_carrier_flag = mem_vazuran_destroyed_planet
								}
							}
						}
					}
				}
			}
			1 = {
			}
		}
	}
}

# Checks if Vazuran country (global event target) exists
# Creates the country and saves it if it does not
create_mem_vazuran_country = {
	if = {
		limit = {
			NOT = {
				exists = event_target:mem_vazurans_country
			}
		}
		create_species = {
			name = "Vazuran"
			plural = "Vazurans"
			class = ART
			portrait = mem_Vazu_crisis
			traits = {
				trait = trait_organic
				trait = trait_erudite
				trait = trait_natural_physicists
				trait = trait_venerable
				trait = trait_slow_breeders
				trait = trait_weak
			}
			extra_trait_points = 3
		}
		create_country = {
			name = "Vazuran Hegemony"
			name_list = "ART2"
			type = mem_vazurans
			
			ignore_initial_colony_error = yes
			authority = auth_dictatorial
			civics = {
				civic = civic_warrior_culture
				civic = civic_barbaric_despoilers
			}
			room = mem_vazuran_1_room
			auto_delete = no
			origin = origin_mem_ancestors_grudge_advanced_start
			species = last_created_species
			# pls let me have this change :)
			# we dont have an fanatic authoritarian FE and vazurans are no genocidal maniacs
			# just a little mad
			ethos = {
				ethic = ethic_militarist
				ethic = ethic_fanatic_authoritarian
			}
			flag = {
				icon = {
					category = "domination"
					file = "domination_21.dds"
				}
				background = {
					category = "backgrounds"
					file = "00_solid.dds"
				}
				colors = {
					"black"
					"dark_purple"
					"null"
					"null"
				}
			}
			effect = {
				set_country_flag = mem_vazurans_country
				set_country_flag = revengeful_patron # bombard rule
			}
		}
		last_created_country = {
			set_graphical_culture = mem_ancient_02
			save_global_event_target_as = mem_vazurans_country
		}
		every_country = {
			limit = {
				OR = {
					is_country_type = default
					is_country_type = fallen_empire
				}
			}
			event_target:mem_vazurans_country = {
				set_faction_hostility = {
					target = prev
					set_hostile = yes
					set_neutral = no
					set_friendly = no
				}
			}
		}
		# Should not attack Enclaves, Guardians, ai empires or the surveyor probe
		# has_country_flag = ne_country for enclaves from new enclaves mod
		# has_country_flag = mem_vazurans_neutral is for other mods to use for special countries that should not be harmed
		every_country = {
			limit = {
				OR = {
					is_country_type = enclave
					is_country_type = primitive
					has_country_flag = mem_surveyor_country
					has_country_flag = ne_country
					has_country_flag = mem_vazurans_neutral
					is_guardian_country = yes
					# AND = {
					# 	has_global_flag = mem_vazurans_neutral
					# 	is_country_type = default
					# 	is_ai = yes
					# }
				}
			}
			event_target:mem_vazurans_country = {
				set_faction_hostility = {
					target = prev
					set_hostile = no
					set_neutral = yes
					set_friendly = no
				}
			}
		}
	}
}

mem_vazurans_remove_stance_modifier = {
	if = {
		limit = {
			has_modifier = mem_vazurans_stance_bunker_down
		}
		remove_modifier = mem_vazurans_stance_bunker_down
	}
	else_if = {
		limit = {
			has_modifier = mem_vazurans_stance_provoke
		}
		remove_modifier = mem_vazurans_stance_provoke
	}
	else_if = {
		limit = {
			has_modifier = mem_vazurans_stance_tribute
		}
		remove_modifier = mem_vazurans_stance_tribute
		hidden_effect = {
			random_list = {
				# chance to obtain tech
				50 = {
				}
				50 = {
					country_event = {
						id = mem_vazurans.1220
						days = 30
					}
				}
			}
		}
	}
	if = {
		limit = {
			has_special_project = MEM_VAZURANS_DAMPER_FIELD_GENERATOR_INVESTIGATION
		}
		abort_special_project = {
			type = MEM_VAZURANS_DAMPER_FIELD_GENERATOR_INVESTIGATION
		}
	}
	if = {
		limit = {
			has_special_project = MEM_VAZURANS_RIFT_INVESTIGATION
		}
		abort_special_project = {
			type = MEM_VAZURANS_RIFT_INVESTIGATION
		}
	}
	end_event_chain = mem_vazuran_menace_chain
}

mem_vazurans_yoink_this_system = {
	if = {
		limit = {
			exists = owner
		}
		owner = {
			save_event_target_as = mem_vazuran_ai_to_get_fucked
		}
		starbase = {
			set_owner = event_target:mem_vazurans_country
		}
		mem_vazurans_create_home_system_defences = yes
		while = {
			count = 3
			random_system_planet = {
				create_mem_vazuran_fleet = yes
			}
		}
		if = {
			limit = {
				event_target:mem_vazurans_country = {
					is_country_type = awakened_fallen_empire
				}
			}
			event_target:mem_vazurans_country = {
				declare_war = {
					target = event_target:mem_vazuran_ai_to_get_fucked
					attacker_war_goal = wg_fe_cleanse_border_worlds
				}
			}
		}
	}
	else = {
		mem_vazurans_create_home_system_defences = yes
		while = {
			count = 3
			random_system_planet = {
				create_mem_vazuran_fleet = yes
			}
		}
	}
	save_global_event_target_as = mem_vazurans_yoinked_system
	spawn_megastructure = {
		type = mem_vazuran_gateway_endpoint
		planet = system_star
	}
}

mem_vazuran_raiding_fleet_spawn = {
	IF = {
		# weakest raiding fleet if not having researched destroyers yet
		limit = {
			ROOT = {
				NOT = {
					has_technology = tech_destroyers
				}
			}
		}
		create_mem_vazuran_corvette_stage_fleet = yes
	}
	IF = {
		# raiding fleet if not having researched cruisers yet
		limit = {
			ROOT = {
				has_technology = tech_destroyers
				NOT = {
					has_technology = tech_cruisers
				}
			}
		}
		create_mem_vazuran_destroyer_stage_fleet = yes
	}
	IF = {
		# raiding fleet if not having researched battleships yet
		limit = {
			ROOT = {
				has_technology = tech_cruisers
				NOT = {
