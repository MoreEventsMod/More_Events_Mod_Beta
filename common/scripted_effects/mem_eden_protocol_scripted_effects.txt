# add situtation progress
mem_eden_protocol_add_protocol_progress = {
	if = {
		limit = {
			any_situation = {
				is_situation_type = situation_mem_eden_protocol_protocolswill
			}
		}
		random_situation = {
			limit = {
				is_situation_type = situation_mem_eden_protocol_protocolswill
			}
			add_situation_progress = $AMOUNT$
		}
	}
	# this is sad actually^^
	else = {
		add_resource = {
			unity = 100
			mult = $AMOUNT$
		}
	}
}

# this is on top of the actual terraforming
mem_eden_protocol_on_terraformed = {
	space_owner = {
		mem_eden_protocol_add_protocol_progress = {
			AMOUNT = 25
		}
	}
	if = {
		limit = {
			habitable_planet_not_urban = yes
			NOT = {
				is_planet_class = pc_gaia
			}
		}
		random_list = {
			85 = {}

				change_pc = pc_nuked
			}

			type = d_active_volcano
		}
		add_blocker = {
			type = d_active_volcano
		}
		add_blocker = {
			type = d_active_volcano

# create random pre ftl on scoped planet
mem_eden_protocol_create_pre_ftl = {
	generate_very_early_pre_ftls_on_planet = yes
	last_created_country = {
		establish_communications = root.space_owner
		if = {
			limit = {
				NOT = {
					any_situation = {
						is_situation_type = pre_ftl_tech_progress_situation
					}
				}
			}
			start_situation = {
				type = pre_ftl_tech_progress_situation
			}
		}
		root = {
			set_owner = prev
		}
	}
	##################################################################
	# planetary diversity effect to prevent dummy habitability traits
	pd_remove_dummy_traits = yes
	##################################################################
	every_owned_pop_group = {
		limit = {
			has_trait = trait_machine_unit
		}
		kill_pop_group = {
			pop_group = this
			percentage = 100
		}
	}	
}
