mem_mortis_spawn_spore_ships = {
	while = {
		count = $COUNT$
		create_ship = {
			fleet_cache_recalc = no
			design = NAME_mem_mortis_spores
			effect = {
				set_name = MEM_MORTIS_SPORE
			}
		}
	}
}

	# create_ambient_object = {
	# 	type = mem_mortis_01_entity_ambient
	# 	location = this
	# }
	# last_created_ambient_object = {
	# 	set_location = {

	# 		distance = 0
	# 		angle = 0
	# 	}
	# 	set_ambient_object_flag = mem_mortis_01_entity_ambient@prev
	# }
	# change_pc = pc_infested
	if = {
		limit = {
			is_artificial = no
		}
		change_pc = pc_tropical
		set_planet_entity = {
			entity = mem_mortis_wooden_planet_entity
			picture = pc_infested
		}
	}
	add_modifier = {
		modifier = mem_mortis_shroomed_planet
	}
	if = {
		limit = {
			exists = event_target:mem_mortis_create_spore_country
		}
		event_target:mem_mortis_create_spore_country = {
			change_variable = {
				which = mem_mortis_create_spore_country_psi_power
				value = 1
			}
			mem_mortis_update_psi_power = yes
		}
		if = {
			limit = {
				host_has_dlc = "Shadows of the Shroud"
			}
			solar_system = {
				spawn_psionic_aura = {
					owner = root.owner
					type = mem_mortis_aura
					update_type = yes
				}
			}
		}
	}
}

mem_mortis_update_psi_power = {
	remove_modifier = mem_mortis_psi_power_modifier
	add_modifier = {
		modifier = mem_mortis_psi_power_modifier
		mult = mem_mortis_create_spore_country_psi_power
	}
}

mem_mortis_create_hunter_fleet = {
	if = {
		limit = {
			exists = event_target:mem_mortis_create_spore_country
		}
		create_fleet = {
			name = "NAME_mem_mortis_hunter"
			settings = {
				spawn_debris = no
				uses_naval_capacity = no
			}
			effect = {
				set_owner = event_target:mem_mortis_create_spore_country
				# set_fleet_bombardment_stance = mem_mortis_fungi_bombardment
				mem_mortis_spawn_spore_ships = {
					COUNT = 20
				}
				set_location = {
					target = $LOCATION$
					distance = 100
					angle = random
				}
			}
		}
	}
}
