decision_mem_forge_world = {
	owned_planets_only = yes
	sound = event_alien_nature
	icon = decision_city_ruins
	resources = {
		category = decisions
	}
	potential = {
		habitable_structure = no
		NOT = {
			has_modifier = mem_vazurans_forge_planet
		}
		owner = {
			has_tradition = tr_mem_vazurans_finish
			NOT = {
				has_country_flag = decision_mem_forge_world
			}
		}
	}
	allow = {
		is_special_colony_type = no
		# not sure about this
		custom_tooltip = {
			fail_text = REQUIRES_MEM_VAZURAN_DECISION_DISTRICT_SET
			#OR = {
				planet_is_ecu_equivalent = yes
				# planet_is_ring_world_equivalent = yes
			#}
		}
	}
	on_queued = {
		hidden_effect = {
			owner = {
				set_country_flag = decision_mem_forge_world
			}
		}
	}
	on_unqueued = {
		hidden_effect = {
			owner = {
				remove_country_flag = decision_mem_forge_world
			}
		}
	}
	effect = {
		add_modifier = {
			modifier = mem_vazurans_forge_planet
		}
	}
	ai_weight = {
		weight = 1
		modifier = {
			OR = {
				has_designation = col_foundry
				has_designation = col_ecu_foundry
				has_designation = col_hiv_foundry
				has_designation = col_mac_foundry
				has_designation = col_capital_foundry
			}
			factor = 200
		}
	}
}
