decision_build_macro_computer = {
	owned_planets_only = yes
	sound = event_administrative_work
	icon = decision_engineering_research
	allow = {
	}
	enactment_time = 3600
	resources = {
		category = decisions
		cost = {
			energy = 7500
			alloys = 5000
		}
	}
	potential = {
		exists = event_target:prof_mem_vir_leader
		owner = {
			has_country_flag = macro_computer_unlocked
			NOT = {
				has_country_flag = building_macro_computer
			}
		}
	}
	on_queued = {
		# only one
		owner = {
			set_country_flag = building_macro_computer
		}
	}
	on_unqueued = {
		owner = {
			remove_country_flag = building_macro_computer
		}
	}
	effect = {
		add_modifier = {
			modifier = mem_vir_pc_macro_computer
		}
		custom_tooltip = decision_build_macro_computer_tooltip
		hidden_effect = {
			planet_event = {
				id = mem_virprof.1011
			}
		}
	}
}
