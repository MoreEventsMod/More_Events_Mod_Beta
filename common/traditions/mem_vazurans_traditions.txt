	possible = {
		OR = {
			custom_tooltip = {
				fail_text = REQUIRES_MEM_OBTAINED_VAZURAN_TECH
				has_country_flag = mem_vazurans_traditions_unlocked
			}
			has_technology = tech_mem_vazuran_xentronium_armor
		}
	}
	modifier = {
		planet_engineers_engineering_research_produces_add = 0.3
	}
	on_enabled = {
		hidden_effect = {
			if = {
				limit = {
					is_country_type = default
					exists = event_target:mem_vazurans_country
					NOT = {
						is_same_value = event_target:mem_vazurans_country
					}
				}
				event_target:mem_vazurans_country = {
					add_opinion_modifier = {
						who = root
						modifier = mem_vazuran_opinion_immitators
					}
				}
			}
		}
	}
	unlocks_agenda = agenda_mem_vazurans_march_of_progress	# unlocks vazuran tech bit after bit
}
