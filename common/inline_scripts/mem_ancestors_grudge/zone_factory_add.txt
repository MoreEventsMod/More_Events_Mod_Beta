### Regular Empires
# Planets
triggered_district_planet_modifier = {
	potential = {
		exists = owner
		owner = {
			is_gestalt = no
		}
		NOR = {
			is_planet_class = pc_city
			is_planet_class = pc_machine
			is_planet_class = pc_hive
		}
	}
	job_mem_ancestors_grudge_runesmith_add = $AMOUNT$
}
