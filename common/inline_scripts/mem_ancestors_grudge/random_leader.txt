if = {
	limit = {		
		exists = event_target:mem_ancestors_grudge_planet
		event_target:mem_ancestors_grudge_planet = {
			exists = leader
			leader = {
				leader_class = $LEADER$
				NOR = {					
					has_trait = trait_imperial_heir
					is_same_value = event_target:leader_1
					is_same_value = event_target:leader_2
					is_same_value = event_target:leader_3					
				}
			}
		}
	}
	event_target:mem_ancestors_grudge_planet = {
		leader = {
			save_event_target_as = leader_forge_leader
		}

	}
}


random_owned_leader = {
	limit = {
		leader_class = $LEADER$
		NOR = {
			mem_ancestors_grudge_has_relic_trait = yes
			is_ruler = yes
			has_trait = trait_imperial_heir
			is_same_value = event_target:leader_forge_leader
		}
	}
	save_event_target_as = leader_1
}

				has_trait = trait_imperial_heir

			}

	}
