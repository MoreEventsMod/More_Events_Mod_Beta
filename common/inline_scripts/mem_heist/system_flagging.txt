random_system = {
	limit = {
		has_star_flag = $SYSTEM_FLAG$
		NOT = {
			has_star_flag = mem_heist_potential_system
		}
	}
	set_star_flag = mem_heist_potential_system
	set_star_flag = mem_heist_$SYSTEM_RESOURCE$	
	event_target:global_event_country = {
		change_variable = {
			which = ap_mem_heist
			value = 1
		}
	}		
}
