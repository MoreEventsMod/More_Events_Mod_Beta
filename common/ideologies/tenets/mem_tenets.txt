tenet_mem_eden = {
	priority = high
	potential = {
		has_origin = origin_mem_eden_protocol
	}
	source_civic = origin_mem_eden_protocol
	on_unlock = {
		custom_tooltip = tenet_mem_eden_effect_tt		# gives passive observation bonus
		hidden_effect = {
			change_variable = {
				which = observation_post_insight
				value = 1
			}
		}
	}
	on_lock = {
		hidden_effect = {
			change_variable = {
				which = observation_post_insight
				value = -1
			}
		}
	}
	first_tier = 1
	tier = {
		triggered_country_modifier = {
			potential = {
				always = yes
			}
			station_observers_produces_mult = 0.25
		}
	}
	tier = {
		triggered_country_modifier = {
			potential = {
				always = yes
			}
			station_observers_produces_mult = 0.5
		}
	}
	tier = {
		triggered_country_modifier = {
			potential = {
				always = yes
			}
			station_observers_produces_mult = 0.75
		}
	}
}
