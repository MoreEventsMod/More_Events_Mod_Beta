# Fired whenever a new owner is set for a planet, following a war or through trade
# This = Planet scope
# From = Country scope (new owner)
# FromFrom = Country scope (previous owner, if any)

	events = {
		mem_ancestors_grudge.200		# lost forge
		mem_ancestors_grudge.210		# got forge back
		mem_ancestors_grudge_durin.7 	# kazad dum gained
	}
}

# Triggers country_event for the attacker upon victory (before controller is switched)

# From = country, planet owner
# FromFrom = planet

	events = {
		mem_ancestors_grudge.505 # relic stealing chance for hive origin
	}	
}
