r_mem_ancestors_grudge_modern_art = {
	activation_duration = 3600
	portrait = GFX_r_mem_ancestors_grudge_modern_art
	sound = "relic_activation_miniature_galaxy"
	resources = {
		category = relics
		# Activation cost
		cost = {
			influence = 300
		}
	}
	ai_weight = {
		weight = 20000
	}
	can_be_stolen = no
	score = 0
	triggered_country_modifier = {
		potential = {
			always = yes
		}
		custom_tooltip = r_mem_ancestors_grudge_cube_effect_tt_passive
	}
	triggered_country_modifier = {
		potential = {
			always = yes
		}
		country_unity_produces_mult = -0.01
	}
	active_effect = {
		custom_tooltip = r_mem_ancestors_grudge_cube_effect_tt_active
		hidden_effect = {
			set_variable = {
				which = mem_ancestors_grudge_ma_amount
				value = 0
			}
			export_resource_stockpile_to_variable = {
				resource = minor_artifacts
				variable = mem_ancestors_grudge_ma_amount
			}
			if = {
				limit = {
					mid_game_years_passed >= 0
				}
				multiply_variable = {
					which = mem_ancestors_grudge_ma_amount
					value = 2
				}
			}
			if = {
				limit = {
					end_game_years_passed >= 0
				}
				multiply_variable = {
					which = mem_ancestors_grudge_ma_amount
					value = 2
				}
			}
			if = {
				limit = {
					is_variable_set = mem_ancestors_grudge_ma_amount
				}
				add_resource = {
					minor_artifacts = -1
					mult = mem_ancestors_grudge_ma_amount
				}
				add_resource = {
					unity = 10
					mult = mem_ancestors_grudge_ma_amount
				}
			}
			clear_variable = mem_ancestors_grudge_ma_amount
			change_variable = {
				which = mem_ancestors_grudge_activated_relics_counter
				value = 1
			}
		}
		#custom_tooltip = r_mem_ancestors_grudge_anvil_effect_tt_active
		custom_tooltip = relic_triumph_cooldown
		hidden_effect = {
			add_modifier = {
				modifier = "relic_activation_cooldown"
				days = 3600
			}
		}
	}
	# Possible check for activation
	possible = {
		inline_script = {
			script = relics/activation_checks
			RELIC = r_mem_ancestors_grudge_modern_art
		}
	}
}
