@activation_cost = 3000
@triumph_duration = 3600
@triumph_duration_short = 1800
@triumph_duration_veryshort = 900
# per planet zombies
r_mem_mortis_undead_scale = {
	activation_duration = 0
	portrait = "GFX_mem_mortis_undead_scale"
	sound = "relic_activation_miniature_galaxy"
	resources = {
		category = relics
		# Activation cost
		cost = {
			unity = 3000
			trigger = {
				is_spiritualist = no
			}
		}
		cost = {
			influence = 300
			trigger = {
				is_spiritualist = yes
			}
		}
	}
	ai_weight = {
		weight = 20
	}
	can_be_stolen = yes
	triggered_country_modifier = {
		potential = {
			always = yes
		}
		planet_farmers_exotic_gases_produces_add = 0.1
		planet_farmers_food_produces_add = 0.5
		planet_farmers_minerals_upkeep_add = 0.5
	}
	score = 500
	active_effect = {
		hidden_effect = {
			if = {
				limit = {
					OR = {
						# all of these cant use the zombie create effect
						is_tankbound_empire = no
						is_machine_empire = no
						is_hive_empire = no
						owner_main_species = {
							is_robotic = no
							is_machine_species = yes
						}
					}
				}
				every_owned_planet = {
					create_zombie_pop_group = {
						SIZE = prev.value:country_mem_assigned_farmers
					}
					custom_tooltip = r_mem_mortis_undead_scale_effect
				}
			}
			else_if = {
				limit = {
					has_monthly_income = {
						value > 0
						resource = exotic_gases
					}
				}
				add_monthly_resource_mult = {
					resource = exotic_gases
					value = 12
				}
			}
			else = {
				add_resource = {
					exotic_gases = 5000
				}
			}
		}
		custom_tooltip = relic_triumph_cooldown
		hidden_effect = {
			add_modifier = {
				modifier = "relic_activation_cooldown"
				days = 3600
			}
		}
	}
	# Possible check for activation
	possible = {
		custom_tooltip = {
			fail_text = "requires_relic_no_cooldown"
			NOT = {
				has_modifier = relic_activation_cooldown
			}
		}
	}
}
