#override for vanilla shield generator
building_planetary_shield_generator = {
	base_buildtime = @b3_time
	planet_limit = 1
	position_priority = 100
	can_be_ruined = no
	can_build = no
	category = army

	building_sets = {
		government
		fortress
	}

	potential = {
		exists = owner
		owner = { 
			is_wilderness_empire = no
			is_nomadic = no
		} # see 'building_ozone_thickener'
	}
	allow = {
		has_upgraded_capital = yes
		if = {
			limit = { exists = orbital_defence }
			orbital_defence = {
				exists = starbase
				starbase = {
					NOR = {
						has_starbase_building = ring_orbital_shield_generator
						is_starbase_building_building = ring_orbital_shield_generator
					}
				}
			}
		}
	}

	prerequisites = {
		#"tech_planetary_shield_generator"
	}
	show_tech_unlock_if = {
		is_wilderness_empire = no
		is_nomadic = no
	}

	planet_modifier = {
		planet_orbital_bombardment_damage = -0.5
	}
	triggered_planet_modifier = {
		potential = { has_cosmic_storms_dlc = yes }
		planet_storm_devastation_mult = -0.5
	}

	inline_script = {
		script = cosmic_storms/StormBuildingModifiers
	}

	inline_script = { # on_built & on_destroy behaviors
		script = cosmic_storms/PlanetaryShieldOnBuiltDestroy
	}

	resources = {
		category = planet_buildings
		inline_script = {
			script = buildings/nomadic_cost_switcher
			REGULAR_RESOURCE = minerals
			NOMADIC_RESOURCE = volatile_motes
			COST = @b2_minerals
			NOMADIC_COST_MULT = @nomadic_minerals_to_strategic_ratio
		}
		upkeep = {
			energy = @b2_upkeep
		}
	}

	custom_storm_ai_weight = {
		weight = 0
		modifier = {
			add = 5
			has_designation = col_fortress
		}
		modifier = {
			add = 5
			AND = {
				exists = owner
				solar_system? = {
					exists = owner
					owner = { num_cosmic_storms_encountered >= 2 }
				}
			}
		}
	}
}

# mem shield generator
building_mem_planetary_shield_generator = {
	icon = "building_planetary_shield_generator"
	base_buildtime = @b3_time
	planet_limit = 1
	position_priority = 100
	can_be_ruined = no
	category = army
	building_sets = {
		government
		fortress
	}
	potential = {
		exists = owner
		owner = { 
			is_wilderness_empire = no
			is_nomadic = no
		}		# see 'building_ozone_thickener'
		planet = {
			# for other mods that may make asteroids colonizable
			NOT = {
				is_planet_class = pc_asteroid
			}
		}
	}
	allow = {
		has_upgraded_capital = yes
		mem_has_shield_building = no
	}
	prerequisites = {
		"tech_planetary_shield_generator"
	}
	show_tech_unlock_if = {
		is_wilderness_empire = no
		is_nomadic = no
	}
	planet_modifier = {
		planet_orbital_bombardment_damage = -0.5
	}
	triggered_planet_modifier = {
		potential = { has_cosmic_storms_dlc = yes }
		planet_storm_devastation_mult = -0.5
	}

	inline_script = {
		script = cosmic_storms/StormBuildingModifiers
	}

	# inline_script = { # on_built & on_destroy behaviors
	# 	script = cosmic_storms/PlanetaryShieldOnBuiltDestroy
	# }
	# need to make changes to on_destroy here
	on_built = {
		if = {
			limit = {
				has_cosmic_storms_dlc = yes
			}
			solar_system = {
				set_star_flag = storm_no_negative_effects
				remove_modifier = storm_base_modifier
			}
		}
	}
	on_destroy = {
		if = {
			limit = {
				has_cosmic_storms_dlc = yes
			}
			owner = {
				set_country_flag = shield_on_destroy
			}			#an action when destroyed
			solar_system = {
				remove_star_flag = storm_no_negative_effects
			}
		}
		carrier_event = {
			id = mem_planetary_shields.2
		}
	}
	resources = {
		category = planet_buildings
		cost = {
			minerals = @b2_minerals
		}
		upkeep = {
			energy = @b2_upkeep
		}
	}
	upgrades = {
		building_mem_planetary_shield_generator_heavy
	}
	ai_weight = {
		weight = 10
		modifier = {
			factor = 0
			pop_amount <= 1000
		}
		modifier = {
			factor = 10
			pop_amount >= 5000
		}
		modifier = {
			factor = 10
			num_buildings = {
				type = building_fortress
				value > 0
			}
		}
	}
	custom_storm_ai_weight = {
		weight = 0
		modifier = {
			add = 5
			has_designation = col_fortress
		}
		modifier = {
			add = 5
			AND = {
				exists = owner
				solar_system? = {
					exists = owner
					owner = { num_cosmic_storms_encountered >= 2 }
				}
			}
		}
	}
}
