building_mem_eden_protocol_1 = {
	icon = building_mem_eden_protocol_seeder_1
	base_buildtime = @b2_time
	planet_limit = 1
	category = unity
	building_sets = {
		government
		unity
	}
	can_be_disabled = no
	upgrades = {
		building_mem_eden_protocol_2
	}
	resources = {
		category = planet_buildings
		cost = {
			minerals = @b1_minerals
		}
		upkeep = {
			energy = @b1_upkeep
		}
	}
	potential = {
		is_planet_class = pc_habitat
		exists = owner
		owner = {
			has_origin = origin_mem_eden_protocol
			NOT = {
				has_valid_civic = civic_machine_assimilator
			}
		}
	}
	allow = {
		custom_tooltip = {
			fail_text = "mem_eden_protocol_building_requirement_1"
			solar_system = {
				any_system_planet = {
					OR = {
						is_terraforming_candidate = yes
						AND = {
							OR = {
								habitable_planet_not_urban = yes
								is_planet_class = pc_nuked
							}
							OR = {
								NOT = {
									exists = owner
								}
								AND = {
									exists = owner
									owner = {
										is_country_type = primitive
									}
								}
							}
						}
					}
				}
			}
		}
	}
	convert_to = {
		building_mem_eden_protocol_da_1
	}
	destroy_trigger = {
		NAND = {
			exists = owner
			owner = {
				has_origin = origin_mem_eden_protocol
			}
		}
	}
	on_destroy = {
		solar_system = {
			every_system_planet = {
				limit = {
					exists = owner
					owner = {
						is_country_type = primitive
						OR = {
							has_country_flag = machine_age
							has_country_flag = atomic_age
							has_country_flag = early_space_age
						}
					}
				}
				owner = {
					remove_country_flag =  tech_frozen
					
				}
			}
		}
	}
	triggered_desc = {
		text = mem_eden_protocol_same_system
	}
	triggered_desc = {
		text = mem_eden_protocol_start_terraforming
	}
	triggered_desc = {
		text = mem_eden_protocol_more_jobs_per_cared_pre_ftl
	}	
	planet_modifier = {
		job_mem_environmental_adjuster_drone_add = 100
	}
	triggered_planet_modifier = {
		job_mem_caretaker_drone_add = 100
		multiplier = solar_system.value:mem_eden_protocol_mem_eden_pre_ftl_in_system
	}
	ai_weight = {
		weight = 1000
	}
	#Always build the pop to work it can come later.
}
