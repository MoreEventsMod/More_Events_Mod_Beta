mem_is_dangerous_system = {
	OR = {
		any_ship_in_system = {
			is_ship_size = pirate_station
		}
		has_star_flag = guardian
		has_star_flag = hostile_system
		has_star_flag = marauder_system
	}
}

mem_is_commie = {
	OR = {
		has_valid_civic = civic_worker_coop
		has_valid_civic = civic_shared_burden
	}
}

mem_has_defender_otg = {
	OR = {
		has_ascension_perk = ap_defender_of_the_galaxy
		has_ascension_perk = ap_defender_of_the_galaxy_nomads
	}
}

mem_loves_fire = {
	OR = {
		has_valid_civic = civic_scorched_earth
		has_valid_civic = civic_hive_scorched_earth
		has_valid_civic = civic_pyrolatry
		owner_main_species = {
			is_species_class = INF
		}
		has_ascension_perk = ap_galactic_hyperthermia
	}
}
