### checks if mem_caretaker is pleased OR content [country scope]
mem_caretaker_mood_is_not_bad_check = {
	OR = {
		mem_caretaker_mood_is_content_check = yes
		mem_caretaker_mood_is_pleased_check = yes
	}
}

	}

### check if planet has mem_caretaker core or mem_caretaker hub [planet scope]
mem_caretaker_presence_on_planet = {
	has_building = building_mem_caretaker_outreach
}

mem_caretaker_tech_access = {
	AND = {
		has_origin = origin_mem_caretaker
		OR = {
			has_country_flag = mem_caretaker_reformed_mem_monitor_order
			NOT = {
				has_country_flag = mem_caretaker_break_up
			}
		}
	}
}

mem_caretaker_tech_access_swap = {
	AND = {
		exists = owner
		owner = {
			# mem_caretaker_tech_access = no
			has_country_flag = mem_caretaker_reformed_mem_monitor_order			# this is also used for when we want to swap
		}
	}
}
