mem_vazuran_is_yoinkable_system = {
	is_bottleneck_system = no
	OR = {
		NOT = {
			exists = owner
		}
		AND = {
			exists = owner
			owner = {
				is_country_type = default
				is_ai = yes
				NOT = {
					is_subject = yes					# we dont wanna yoink from subjects
				}
			}
		}
	}
}

mem_vazuran_was_raid_target = {
	OR = {
		has_modifier = mem_vazurans_stance_bunker_down
		has_modifier = mem_vazurans_stance_provoke
		has_modifier = mem_vazurans_stance_tribute
		has_special_project = MEM_VAZURANS_DAMPER_FIELD_GENERATOR_INVESTIGATION
		has_event_chain = mem_vazuran_menace_chain
	}
	is_country_type = default
}
