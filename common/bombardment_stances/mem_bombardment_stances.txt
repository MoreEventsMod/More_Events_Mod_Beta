# Copy of armageddon stance for faction benefit.
mem_armageddon = {
	trigger = {
		owner = { 
			OR = {
                is_country_type = mem_station_smasher
                is_country_type = mem_neutral_station_smasher
            }		
		}
	}
	
	default = no
	
	stop_when_armies_dead = no
	stop_when_ground_combat = no
	abduct_pops = no

	planet_damage = 1.5
	army_damage = 2.0

	kill_pop_chance = {
		base = 1.0
	}
	min_pops_to_kill_pop = 0
	kill_pop_amount = { min = 0 max = 200 }
	
	# root = fleet
	# from = planet
	ai_weight = {
		weight = 0
	}
}

mem_eden_protocol_devolve = {
	trigger = {
		owner = {
			OR = {
				has_relic = r_mem_eden_protocol_devolve_relic
				AND = {
					owner = { is_country_type = enclave_mercenary }
					root.controller = {
						has_relic = r_mem_eden_protocol_devolve_relic
					}
				}
			}			
		}
	}

	default = no

	stop_when_armies_dead = no
	stop_when_ground_combat = no
	abduct_pops = no

	planet_damage = 0.8 
	army_damage = 1

	kill_pop_chance = {
		base = 1.0
	}
	min_pops_to_kill_pop = 0
	kill_pop_amount = { min = 0 max = 200 }

	# root = fleet
	# from = planet
	ai_weight = {
		weight = 10
		modifier = {
			factor = 0
			exists = from
			from = {
				owner = { is_country_type = primitive }
			}
		}
	}
}

mem_vazurans_plundering = {
	trigger = {
		exists = owner
		OR = {
			owner = { has_tradition = tr_mem_vazurans_4 }
			owner = { is_country_type = mem_vazurans }
		}
		
		NOT = {
			any_owned_ship = {
				has_ship_flag = behemoth_hatchling_ship
			}
		}
	}

	default = no

	stop_when_armies_dead = no
	stop_when_ground_combat = no

	planet_damage = 0.15
	army_damage = 0.35
	
	kill_pop_chance = {
		base = 0.1
	}
	abduct_pops = yes
	min_pops_to_kill_pop = 10
	kill_pop_amount = { min = 10 max = 100 }

	# root = fleet
	# from = planet
	ai_weight = {
		weight = 8
		modifier = { # AI doesn't plunder backwater planets with no resources
			factor = 0
			exists = from
			from = {
				has_upgraded_capital = no
			}
		}
	}
}
